ALPHA TITANS - OWNER SETUP (ONE TIME, ABOUT 10 MINUTES)

WHERE TO DO IT
  1. Open https://console.cloud.google.com and choose project
     samriddhi-production.
  2. Click the Cloud Shell icon (">_", top right).
  3. Paste the commands below. You must be signed in as a project Owner.
  Nothing to install.

FILL IN 2 VALUES FIRST: L4_VM and ZONE (the first lines of the script).

AFTER IT RUNS, PLEASE SEND BACK: the L4 VM name, its zone, and "done".


------------------------------------------------------------------------
# values
PROJECT=samriddhi-production
L4_VM=REPLACE_WITH_L4_VM_NAME
ZONE=REPLACE_WITH_ZONE            # for example us-central1-a
REGION=${ZONE%-*}
BUCKET=${PROJECT}-alphatitans-media
ME=akansha.lakhina@mysamriddhi.com
COND='expression=request.time < timestamp("2026-11-21T00:00:00Z"),title=expires-21-nov'

# 1. APIs
gcloud services enable compute.googleapis.com storage.googleapis.com iamcredentials.googleapis.com --project=$PROJECT

# 2. Service accounts (no keys)
gcloud iam service-accounts create at-control --display-name="Alpha Titans control" --project=$PROJECT
gcloud iam service-accounts create at-worker --display-name="Alpha Titans worker" --project=$PROJECT
CTRL=at-control@$PROJECT.iam.gserviceaccount.com
WORK=at-worker@$PROJECT.iam.gserviceaccount.com

# 3. One private bucket; both accounts limited to it
gcloud storage buckets create gs://$BUCKET --project=$PROJECT --location=$REGION --uniform-bucket-level-access --public-access-prevention
gcloud storage buckets add-iam-policy-binding gs://$BUCKET --member=serviceAccount:$CTRL --role=roles/storage.objectAdmin
gcloud storage buckets add-iam-policy-binding gs://$BUCKET --member=serviceAccount:$WORK --role=roles/storage.objectAdmin

# 4. Role to start/stop the L4 only
gcloud iam roles create atWorkerOperator --project=$PROJECT --title="Alpha Titans worker operator" --permissions=compute.instances.get,compute.instances.start,compute.instances.stop --stage=GA
gcloud compute instances add-iam-policy-binding $L4_VM --zone=$ZONE --project=$PROJECT --member=serviceAccount:$CTRL --role=projects/$PROJECT/roles/atWorkerOperator

# 5. L4 uses the worker account (this STOPS the L4; run it only if nobody
#    is using it. Our platform starts it when needed.)
gcloud compute instances stop $L4_VM --zone=$ZONE --project=$PROJECT
gcloud compute instances set-service-account $L4_VM --zone=$ZONE --project=$PROJECT --service-account=$WORK --scopes=cloud-platform
gcloud compute instances add-tags $L4_VM --zone=$ZONE --project=$PROJECT --tags=at-worker

# 6. Control VM (same zone; if the L4 is not on the "default" network,
#    add --network and --subnet of the L4's network)
gcloud compute instances create at-control --project=$PROJECT --zone=$ZONE --machine-type=e2-micro --boot-disk-size=30GB --boot-disk-type=pd-standard --image-family=ubuntu-2204-lts --image-project=ubuntu-os-cloud --service-account=$CTRL --scopes=cloud-platform --tags=at-control

# 7. Firewall (skip a rule if the network already allows it)
gcloud compute firewall-rules create at-worker-to-control --project=$PROJECT --network=default --direction=INGRESS --action=ALLOW --rules=tcp:8000 --source-tags=at-worker --target-tags=at-control
gcloud compute firewall-rules create at-iap-ssh --project=$PROJECT --network=default --direction=INGRESS --action=ALLOW --rules=tcp:22 --source-ranges=35.235.240.0/20 --target-tags=at-control,at-worker

# 8. Access for Akansha (expires 21 Nov 2026)
#    If a command says conditions are not supported, run it again without
#    --condition and remove that access by hand on 21 Nov.
for VM in $L4_VM at-control; do
  gcloud compute instances add-iam-policy-binding $VM --zone=$ZONE --project=$PROJECT --member=user:$ME --role=roles/compute.osAdminLogin --condition="$COND"
done
gcloud compute instances add-iam-policy-binding $L4_VM --zone=$ZONE --project=$PROJECT --member=user:$ME --role=projects/$PROJECT/roles/atWorkerOperator --condition="$COND"
gcloud iam roles create atProjectView --project=$PROJECT --title="Alpha Titans project view" --permissions=compute.projects.get --stage=GA
gcloud projects add-iam-policy-binding $PROJECT --member=user:$ME --role=projects/$PROJECT/roles/atProjectView --condition="$COND"
# tunnel only: logging in to any other VM still needs its own login grant,
# which she does not have
gcloud projects add-iam-policy-binding $PROJECT --member=user:$ME --role=roles/iap.tunnelResourceAccessor --condition="$COND"
for SA in $CTRL $WORK; do
  gcloud iam service-accounts add-iam-policy-binding $SA --project=$PROJECT --member=user:$ME --role=roles/iam.serviceAccountUser --condition="$COND"
done
gcloud storage buckets add-iam-policy-binding gs://$BUCKET --member=user:$ME --role=roles/storage.objectViewer --condition="$COND"
